Universal 1-Line Drop-in Integration

from hyper_reason import wrap_model, SearchPresets
from transformers import AutoModelForCausalLM

# 1. Load your standard PyTorch / HuggingFace model
base_model = AutoModelForCausalLM.from_pretrained("meta-llama/Meta-Llama-3-8B-Instruct")

# 2. Wrap with HyperReason for Zero-Copy FlashKV + MCTS Reasoning
model = wrap_model(base_model, config=SearchPresets.high_accuracy())

# 3. Execute test-time reasoning search
result = model.reason("If 5 workers complete a job in 12 days, how many days for 8 workers?")
print("Boxed Answer:", result["boxed_answer"])
print("VRAM Saved:", result["metrics"]["flash_kv_stats"]["saved_vram_mb"], "MB")
